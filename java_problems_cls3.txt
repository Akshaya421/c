class Main {
   public static void main(String[] args){
    int n=5;
    char ch='a';
    for(int i=1; i<=n; i++){
        for(int j=1; j<=n-i; j++){
         
            System.out.print(" ");
}
        
        for(int k=1; k<=i; k++){
  
           System.out.print(ch+" "); 
           ch++;
           
        } 
        
        for(int l=1; l<=i; l++){
  
           System.out.print(" "); 
           --ch;
        } 
        
        System.out.println(" ");
    } 
    
}

}

    int n=5;
    char ch='a';
    for(int j=0; j<n; j++){
        for(int k=0; k<n-j-1; k++)
            System.out.print(" ");

        for(int i=0; i<j+1; i++){
         
            System.out.print((char)('a'+i));
}
        
        for(int i=j; i>0; i--){
  
           System.out.print((char)('a'+(i-1)));
